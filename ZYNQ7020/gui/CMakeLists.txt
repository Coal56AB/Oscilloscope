cmake_minimum_required(VERSION 3.16)
project(oscill_lcd C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)

option(SCOPE_BUILD_PREVIEW "Build the desktop/SDL frontend" ON)
option(SCOPE_ENABLE_DRM "Enable direct DRM dumb-buffer output and evdev touch" OFF)
option(SCOPE_ENABLE_FRAMEBUFFER "Enable Linux framebuffer output and evdev touch" OFF)
option(SCOPE_ENABLE_ZYNQ "Enable the Linux Zynq capture adapter" OFF)
option(SCOPE_BUILD_BENCHMARKS "Build synthetic capture and renderer benchmarks" ON)
find_package(Threads REQUIRED)
add_library(signal_generator STATIC src/signal_generator.c)
target_include_directories(signal_generator PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src")
if(NOT MSVC)
    target_link_libraries(signal_generator PUBLIC m)
endif()

add_library(oled STATIC src/oled.c src/oled_font.c)
target_include_directories(oled PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src")
add_library(status_model STATIC src/status_model.c)
target_link_libraries(status_model PUBLIC oled)
if(UNIX AND NOT APPLE)
    add_library(status_ipc STATIC src/status_ipc.c)
    target_link_libraries(status_ipc PUBLIC status_model)
    add_library(status_system STATIC src/status_system.c src/linux_storage.c)
    target_link_libraries(status_system PUBLIC status_model m)
    add_executable(oscill-status src/status_main.c src/linux_oled.c src/linux_status_key.c)
    target_link_libraries(oscill-status PRIVATE status_ipc status_system)
    install(TARGETS oscill-status RUNTIME DESTINATION bin)
endif()

add_library(waveform_view STATIC src/waveform_view.c src/signal_filter.c src/dsp_fft.c src/channel_response.c)
target_include_directories(waveform_view PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src")
if(NOT MSVC)
    target_link_libraries(waveform_view PUBLIC m)
endif()

add_library(control_protocol STATIC src/control_protocol.c)
target_include_directories(control_protocol PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src")
add_library(scope_runtime STATIC
    src/scope_thread.c src/capture.c src/capture_processor.c src/capture_file.c
    src/capture_pipeline.c src/capture_history.c src/capture_debug.c src/control_transport.c src/control_serial.c)
target_include_directories(scope_runtime PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src")
target_link_libraries(scope_runtime PUBLIC control_protocol Threads::Threads waveform_view signal_generator)
if(UNIX AND NOT APPLE)
    target_sources(scope_runtime PRIVATE src/linux_serial.c)
endif()
if(WIN32)
    target_compile_definitions(scope_runtime PUBLIC _WIN32_WINNT=0x0602)
endif()
if(NOT MSVC)
    target_link_libraries(scope_runtime PUBLIC m)
endif()
if(SCOPE_ENABLE_ZYNQ)
    if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
        message(FATAL_ERROR "The Zynq capture adapter requires Linux")
    endif()
    target_sources(scope_runtime PRIVATE src/capture_zynq.c src/capture_selftest.c)
    target_compile_definitions(scope_runtime PUBLIC SCOPE_ENABLE_ZYNQ=1)
    if(CMAKE_SYSTEM_PROCESSOR MATCHES "^arm")
        target_compile_options(scope_runtime PRIVATE -O3 -mfpu=neon)
    endif()
endif()

add_library(boot_splash STATIC logo/boot_splash.c)
target_include_directories(boot_splash PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/logo")

add_library(scope_screen STATIC src/scope_screen.c src/rounded_box.c)
target_link_libraries(scope_screen PUBLIC boot_splash waveform_view signal_generator)
target_include_directories(scope_screen PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src")
if(SCOPE_ENABLE_ZYNQ AND CMAKE_SYSTEM_PROCESSOR MATCHES "^arm")
    target_compile_options(scope_screen PRIVATE -O3 -mfpu=neon)
endif()

add_library(scope_renderer STATIC src/scope_renderer.c)
target_link_libraries(scope_renderer PUBLIC scope_screen scope_runtime)

add_library(scope_settings STATIC src/demo_settings.c src/settings_store.c src/settings_session.c)
target_link_libraries(scope_settings PUBLIC scope_screen scope_runtime)

if(SCOPE_ENABLE_ZYNQ)
    add_library(firmware_flash STATIC src/firmware_flash.c)
    target_include_directories(firmware_flash PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src")
    add_executable(oscill-update src/firmware_update.c src/linux_display.c)
    target_link_libraries(oscill-update PRIVATE firmware_flash scope_screen)
    install(TARGETS oscill-update RUNTIME DESTINATION bin)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(scope_status_test src/tests/test_status.c)
    target_link_libraries(scope_status_test PRIVATE status_model)
    add_test(NAME scope_status COMMAND scope_status_test)
    if(UNIX AND NOT APPLE)
        add_executable(scope_status_key_test src/tests/test_status_key.c src/linux_status_key.c)
        target_include_directories(scope_status_key_test PRIVATE src)
        target_link_options(scope_status_key_test PRIVATE
            -Wl,--wrap=glob -Wl,--wrap=globfree -Wl,--wrap=open -Wl,--wrap=close -Wl,--wrap=ioctl)
        add_test(NAME scope_status_key COMMAND scope_status_key_test)
        add_executable(scope_status_linux_test src/tests/test_status_linux.c)
        target_link_libraries(scope_status_linux_test PRIVATE status_ipc status_system)
        add_test(NAME scope_status_linux COMMAND scope_status_linux_test)
    endif()
    add_executable(scope_sim_test
        src/tests/test_demo_signal.c
        src/demo_signal.c
        src/wave_file.c
    )
    target_link_libraries(scope_sim_test PRIVATE scope_screen control_protocol)
    if(NOT MSVC)
        target_link_libraries(scope_sim_test PRIVATE m)
    endif()
    add_test(NAME scope_simulator COMMAND scope_sim_test)
    add_executable(scope_rendering_test src/tests/test_rendering.c src/demo_signal.c)
    target_link_libraries(scope_rendering_test PRIVATE scope_screen control_protocol)
    if(NOT MSVC)
        target_link_libraries(scope_rendering_test PRIVATE m)
    endif()
    add_test(NAME scope_rendering COMMAND scope_rendering_test)
    add_executable(scope_runtime_test src/tests/test_runtime.c)
    target_link_libraries(scope_runtime_test PRIVATE scope_runtime)
    add_test(NAME scope_runtime COMMAND scope_runtime_test)
    add_executable(scope_filter_test src/tests/test_filter.c)
    target_link_libraries(scope_filter_test PRIVATE scope_runtime)
    add_test(NAME scope_filter COMMAND scope_filter_test)
    add_executable(scope_settings_test src/tests/test_settings.c src/demo_signal.c)
    target_link_libraries(scope_settings_test PRIVATE scope_settings)
    add_test(NAME scope_settings COMMAND scope_settings_test)
    add_executable(scope_renderer_test src/tests/test_renderer.c src/demo_signal.c)
    target_link_libraries(scope_renderer_test PRIVATE scope_renderer control_protocol)
    add_test(NAME scope_renderer COMMAND scope_renderer_test)
    add_executable(scope_response_test src/tests/test_response.c src/demo_signal.c)
    target_link_libraries(scope_response_test PRIVATE scope_settings)
    add_test(NAME scope_response COMMAND scope_response_test)
    add_executable(scope_history_test src/tests/test_capture_history.c)
    target_link_libraries(scope_history_test PRIVATE scope_runtime)
    add_test(NAME scope_history COMMAND scope_history_test)
    if(SCOPE_ENABLE_ZYNQ)
        add_executable(scope_firmware_test src/tests/test_firmware_flash.c)
        target_link_libraries(scope_firmware_test PRIVATE firmware_flash)
        target_link_options(scope_firmware_test PRIVATE -Wl,--wrap=open -Wl,--wrap=fopen -Wl,--wrap=ioctl -Wl,--wrap=pread)
        add_test(NAME scope_firmware COMMAND scope_firmware_test)
        add_executable(scope_zynq_test src/tests/test_zynq_capture.c)
        target_link_libraries(scope_zynq_test PRIVATE scope_runtime)
        target_link_options(scope_zynq_test PRIVATE
            -Wl,--wrap=open -Wl,--wrap=close -Wl,--wrap=mmap -Wl,--wrap=munmap
            -Wl,--wrap=poll -Wl,--wrap=ioctl)
        add_test(NAME scope_zynq_capture COMMAND scope_zynq_test)
    endif()
    if(UNIX AND NOT APPLE)
        add_executable(scope_serial_usb_test src/tests/test_serial_usb.c)
        target_link_libraries(scope_serial_usb_test PRIVATE scope_runtime)
        target_link_options(scope_serial_usb_test PRIVATE -Wl,--wrap=linux_serial_next_usb)
        add_test(NAME scope_serial_usb COMMAND scope_serial_usb_test)
    endif()
endif()

if(SCOPE_BUILD_BENCHMARKS)
    add_executable(scope_benchmark src/tests/benchmark.c src/demo_signal.c)
    target_link_libraries(scope_benchmark PRIVATE scope_runtime scope_screen)
endif()

if(SCOPE_BUILD_PREVIEW AND WIN32)
    add_executable(scope_preview WIN32
        src/preview_win32.c
        src/panel_win32.c
        src/demo_signal.c
        src/wave_file.c
    )
    if(MINGW)
        target_compile_definitions(scope_preview PRIVATE _WIN32_WINNT=0x0602)
        target_link_options(scope_preview PRIVATE -municode)
        target_link_libraries(scope_preview PRIVATE m)
    endif()
    target_link_libraries(scope_preview PRIVATE scope_screen scope_renderer scope_settings control_protocol user32 gdi32 shell32)
elseif(SCOPE_BUILD_PREVIEW AND UNIX AND NOT APPLE)
    find_package(SDL2 CONFIG QUIET)
    if(TARGET SDL2::SDL2)
        set(SCOPE_SDL_TARGET SDL2::SDL2)
    else()
        find_package(PkgConfig REQUIRED)
        pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2)
        set(SCOPE_SDL_TARGET PkgConfig::SDL2)
    endif()
    add_executable(scope_preview
        src/preview_sdl.c
        src/linux_storage.c
        src/panel_sdl.c
        src/demo_signal.c
        src/wave_file.c
        src/capture_adapter.c
        src/remote_control.c
    )
    target_link_libraries(scope_preview PRIVATE scope_screen scope_runtime scope_renderer scope_settings status_ipc ${SCOPE_SDL_TARGET} m)
    if(BUILD_TESTING)
        add_executable(scope_frontend_test src/tests/test_sdl_frontend.c src/linux_storage.c
            src/panel_sdl.c src/demo_signal.c src/wave_file.c src/capture_adapter.c src/remote_control.c)
        target_link_libraries(scope_frontend_test PRIVATE scope_screen scope_runtime scope_renderer scope_settings status_ipc ${SCOPE_SDL_TARGET} m)
        if(SCOPE_ENABLE_ZYNQ)
            target_link_options(scope_frontend_test PRIVATE -Wl,--wrap=capture_selftest
                -Wl,--wrap=capture_pipeline_create_zynq -Wl,--wrap=linux_storage_check
                -Wl,--wrap=scope_sleep_ms -Wl,--wrap=SDL_GetTicks)
        endif()
        add_test(NAME scope_linux_frontend COMMAND scope_frontend_test)
    endif()
    if(SCOPE_ENABLE_DRM OR SCOPE_ENABLE_FRAMEBUFFER)
        target_sources(scope_preview PRIVATE src/linux_display.c src/linux_touch.c)
        target_compile_definitions(scope_preview PRIVATE SCOPE_ENABLE_LINUX_DISPLAY=1)
        if(BUILD_TESTING)
            add_executable(scope_touch_test src/tests/test_linux_touch.c src/linux_touch.c)
            target_link_libraries(scope_touch_test PRIVATE ${SCOPE_SDL_TARGET} m)
            target_link_options(scope_touch_test PRIVATE
                -Wl,--wrap=open -Wl,--wrap=read -Wl,--wrap=ioctl -Wl,--wrap=close)
            add_test(NAME scope_linux_touch COMMAND scope_touch_test)
        endif()
    endif()
    if(SCOPE_ENABLE_DRM)
        find_package(PkgConfig REQUIRED)
        pkg_check_modules(DRM REQUIRED IMPORTED_TARGET libdrm)
        target_compile_definitions(scope_preview PRIVATE SCOPE_ENABLE_DRM=1)
        target_link_libraries(scope_preview PRIVATE PkgConfig::DRM)
    endif()
    add_executable(oscill-remote src/remote_client.c)
    install(TARGETS oscill-remote RUNTIME DESTINATION bin)
endif()

if(TARGET scope_preview)
    install(TARGETS scope_preview RUNTIME DESTINATION bin)
endif()
